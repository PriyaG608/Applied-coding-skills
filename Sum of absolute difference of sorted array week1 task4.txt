class Solution {
    public int[] getSumAbsoluteDifferences(int[] nums) {
        int n = nums.length;
        int[] result = new int[n];
        int totalSum = 0;
        for (int num : nums) {
            totalSum += num;
        }
        int prefixSum = 0;

        for (int i = 0; i < n; i++) {
            int x = nums[i];

            int leftSum = x * i - prefixSum;
            int rightSum = (totalSum - prefixSum - x);
            int rightCount = n - i - 1;
            int rightContribution = rightSum - x * rightCount;

            result[i] = leftSum + rightContribution;
            prefixSum += x;
        }

        return result;
    }
}
